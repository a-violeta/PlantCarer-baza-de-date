CREATE TABLE DETINATOR (
    ID_DETINATOR NUMBER PRIMARY KEY,
    NUME_DET VARCHAR2(40) NOT NULL,
    EMAIL_DET VARCHAR2(40) UNIQUE NOT NULL,
    ADRESA_DET VARCHAR2(100) NOT NULL
);

CREATE TABLE PLANTA (
    ID_PLANTA NUMBER PRIMARY KEY,
    SPECIE VARCHAR2(40) NOT NULL,
    DATA_NASTERII DATE DEFAULT SYSDATE,
    STATUS VARCHAR2(10) NOT NULL CHECK (STATUS IN ('ADOPTABILA', 'DETINUTA'))
);

CREATE TABLE SEDIU (
    ID_SEDIU NUMBER PRIMARY KEY,
    ADRESA_SED VARCHAR2(100) NOT NULL
);

CREATE TABLE SERVICIU (
    ID_SERVICIU NUMBER PRIMARY KEY,
    DENUMIRE_SERV VARCHAR2(20) UNIQUE NOT NULL,
    PRET_SERV FLOAT DEFAULT 50.00 CONSTRAINT ck_pret_interval CHECK (PRET_SERV >= 10 AND PRET_SERV <= 100)
);

CREATE TABLE PLANTA_DETINUTA (
    ID_PLANTA NUMBER PRIMARY KEY REFERENCES PLANTA(ID_PLANTA),
    ID_DETINATOR NUMBER REFERENCES DETINATOR(ID_DETINATOR)
);

CREATE TABLE GRADINA_BOTANICA (
    ID_GRADINA NUMBER PRIMARY KEY,
    ID_SEDIU NUMBER REFERENCES SEDIU(ID_SEDIU),
    NUME_GRAD VARCHAR2(40) UNIQUE NOT NULL
);

CREATE TABLE HABITAT (
    ID_HABITAT NUMBER PRIMARY KEY,
    ID_GRADINA NUMBER REFERENCES GRADINA_BOTANICA(ID_GRADINA),
    MEDIU VARCHAR2(40) UNIQUE NOT NULL,
    CAPACITATE NUMBER DEFAULT 1
);

CREATE TABLE PLANTA_SEDIULUI (
    ID_PLANTA NUMBER PRIMARY KEY REFERENCES PLANTA(ID_PLANTA),
    ID_HABITAT NUMBER REFERENCES HABITAT(ID_HABITAT)
);

CREATE TABLE ANGAJAT (
    ID_ANGAJAT NUMBER PRIMARY KEY,
    ID_SEDIU NUMBER REFERENCES SEDIU(ID_SEDIU),
    DATA_ANGAJARE DATE DEFAULT SYSDATE
);

CREATE TABLE INGRIJITOR_DOMICILIU (
    ID_ANGAJAT NUMBER PRIMARY KEY REFERENCES ANGAJAT(ID_ANGAJAT),
    EMAIL_ANG VARCHAR2(40) UNIQUE NOT NULL
);

CREATE TABLE INGRIJITOR_HABITAT (
    ID_ANGAJAT NUMBER PRIMARY KEY REFERENCES ANGAJAT(id_angajat),
    ID_HABITAT NUMBER REFERENCES HABITAT(id_habitat)
);

CREATE TABLE REZERVARE (
    ID_REZERVARE NUMBER PRIMARY KEY,
    ID_PLANTA NUMBER REFERENCES PLANTA_DETINUTA(ID_PLANTA),
    ID_DETINATOR NUMBER REFERENCES DETINATOR(ID_DETINATOR),
    ID_ANGAJAT NUMBER REFERENCES INGRIJITOR_DOMICILIU(ID_ANGAJAT),
    DATA_INCEPUT DATE,
    DATA_FINAL DATE
);

CREATE TABLE REZERVARE_SERVICIU (
    ID_REZERVARE NUMBER,
    ID_SERVICIU NUMBER,
    PRIMARY KEY (ID_REZERVARE, ID_SERVICIU),
    FOREIGN KEY (ID_REZERVARE) REFERENCES REZERVARE(ID_REZERVARE),
    FOREIGN KEY (ID_SERVICIU) REFERENCES SERVICIU(ID_SERVICIU)
);

CREATE TABLE ADOPTIE (
    ID_ADOPTIE NUMBER PRIMARY KEY,
    ID_DETINATOR NUMBER REFERENCES DETINATOR(ID_DETINATOR),
    ID_PLANTA NUMBER REFERENCES PLANTA(ID_PLANTA),
    VALIDARE_ADOPTIE VARCHAR2(3) CHECK (VALIDARE_ADOPTIE IN ('DA', 'NU')),
    DATA_ADOPTIE DATE
);

CREATE TABLE PLASARE (
    ID_PLASARE NUMBER PRIMARY KEY,
    ID_DETINATOR NUMBER REFERENCES DETINATOR(ID_DETINATOR),
    ID_PLANTA NUMBER REFERENCES PLANTA(ID_PLANTA),
    VALIDARE_PLASARE VARCHAR2(3) CHECK (VALIDARE_PLASARE IN ('DA', 'NU')),
    DATA_PLASARE DATE
);

CREATE TABLE BILET_INDIVIDUAL (
    ID_BILET NUMBER PRIMARY KEY,
    ID_GRADINA NUMBER REFERENCES GRADINA_BOTANICA(ID_GRADINA),
    ID_DETINATOR NUMBER REFERENCES DETINATOR(ID_DETINATOR),
    DATA_VIZITEI DATE DEFAULT SYSDATE,
    PLATIT VARCHAR2(3) DEFAULT 'NU' CHECK (PLATIT IN ('DA', 'NU'))
);
